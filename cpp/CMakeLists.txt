cmake_minimum_required(VERSION 3.18)
project(mlnd_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

set(CPLEX_ROOT "$ENV{HOME}/Applications/CPLEX_Studio222" CACHE PATH "CPLEX Optimization Studio root")
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp HINTS /opt/anaconda3/include /opt/homebrew/include)
if(NOT NLOHMANN_JSON_INCLUDE_DIR)
  message(FATAL_ERROR "nlohmann/json.hpp not found; set NLOHMANN_JSON_INCLUDE_DIR")
endif()
if(APPLE)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(CPLEX_PLATFORM arm64_osx)
  else()
    set(CPLEX_PLATFORM x86-64_osx)
  endif()
else()
  set(CPLEX_PLATFORM x86-64_linux)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
foreach(component cplex ilocplex)
  find_library(${component}_LIBRARY NAMES ${component} HINTS "${CPLEX_ROOT}/cplex/lib/${CPLEX_PLATFORM}/static_pic" NO_DEFAULT_PATH REQUIRED)
endforeach()
find_library(concert_LIBRARY NAMES concert HINTS "${CPLEX_ROOT}/concert/lib/${CPLEX_PLATFORM}/static_pic" NO_DEFAULT_PATH REQUIRED)
add_library(mlnd_cplex INTERFACE)
target_include_directories(mlnd_cplex SYSTEM INTERFACE "${CPLEX_ROOT}/cplex/include" "${CPLEX_ROOT}/concert/include")
target_compile_definitions(mlnd_cplex INTERFACE IL_STD)
target_link_libraries(mlnd_cplex INTERFACE ${ilocplex_LIBRARY} ${concert_LIBRARY} ${cplex_LIBRARY} m pthread)
if(APPLE)
  target_link_libraries(mlnd_cplex INTERFACE "-framework CoreFoundation" "-framework IOKit" "-framework Accelerate")
endif()

add_library(mlnd_input STATIC src/input.cpp)
target_include_directories(mlnd_input PUBLIC include)
target_include_directories(mlnd_input SYSTEM PUBLIC "${NLOHMANN_JSON_INCLUDE_DIR}")
target_compile_options(mlnd_input PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(mlnd_input PUBLIC -ffp-contract=off)
if(NOT APPLE)
  target_link_libraries(mlnd_input PUBLIC OpenSSL::Crypto)
endif()

add_executable(mlnd_input_probe tests/input_probe.cpp)
target_link_libraries(mlnd_input_probe PRIVATE mlnd_input)
add_executable(mlnd_progress_probe tests/progress_probe.cpp src/progress.cpp)
target_link_libraries(mlnd_progress_probe PRIVATE mlnd_input)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/path_catalogue.cpp")
  add_executable(mlnd_path_catalogue tests/path_catalogue.cpp src/paths.cpp)
  target_link_libraries(mlnd_path_catalogue PRIVATE mlnd_input)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  file(GLOB MLND_SOLVER_SOURCES CONFIGURE_DEPENDS src/*.cpp)
  list(REMOVE_ITEM MLND_SOLVER_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/input.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  add_library(mlnd_solver STATIC ${MLND_SOLVER_SOURCES})
  target_link_libraries(mlnd_solver PUBLIC mlnd_input mlnd_cplex)
  target_compile_options(mlnd_solver PRIVATE -Wall -Wextra -Wpedantic)
  add_executable(mlnd_cpp src/main.cpp)
  target_link_libraries(mlnd_cpp PRIVATE mlnd_solver)
  foreach(probe origin_assignment fixed_network formulation output benders fixed_design pricing strengthening branch_seed heuristics cut_pool root_diagnostic continuous continuous_matrix bdd bdd_search dual_workspace tight_airline topology_cuts assignment_support_cuts assignment_support_validity branch_joint_cuts branch_joint_validity od_airline od_airline_root conditional_cut conditional_formulation conditional_benders structural_strengthening structural_cg_cut)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/${probe}_probe.cpp")
      add_executable(mlnd_${probe}_probe tests/${probe}_probe.cpp)
      target_link_libraries(mlnd_${probe}_probe PRIVATE mlnd_solver)
    endif()
  endforeach()
endif()
