cmake_minimum_required(VERSION 3.18)
project(mlnd_cpp VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(DEFINED ENV{CPLEX_ROOT})
  set(_CPLEX_ROOT "$ENV{CPLEX_ROOT}")
elseif(APPLE)
  set(_CPLEX_ROOT "$ENV{HOME}/Applications/CPLEX_Studio222")
else()
  set(_CPLEX_ROOT "")
endif()
set(CPLEX_ROOT "${_CPLEX_ROOT}" CACHE PATH "CPLEX Optimization Studio root (required on Windows/Linux)")
if(NOT EXISTS "${CPLEX_ROOT}/cplex/include/ilcplex/ilocplex.h" OR
   NOT EXISTS "${CPLEX_ROOT}/concert/include/ilconcert/iloenv.h")
  message(FATAL_ERROR "Set CPLEX_ROOT to the installed CPLEX Optimization Studio directory (C++/Concert headers required)")
endif()
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp HINTS /opt/anaconda3/include /opt/homebrew/include)
if(NOT NLOHMANN_JSON_INCLUDE_DIR)
  message(FATAL_ERROR "nlohmann/json.hpp not found; set NLOHMANN_JSON_INCLUDE_DIR")
endif()
if(WIN32)
  if(NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows builds require a 64-bit MSVC-compatible toolchain; use Visual Studio with -A x64")
  endif()
  set(CPLEX_PLATFORM x64_windows_msvc14 CACHE STRING "CPLEX installed platform directory")
  set(_CPLEX_LIBRARY_LAYOUT stat_mda)
  # IBM stat_mda uses the release DLL runtime (/MD), including when the
  # application has debugging symbols. Keep every target's STL ABI consistent.
  set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreadedDLL)
  add_compile_definitions(_ITERATOR_DEBUG_LEVEL=0 NOMINMAX WIN32_LEAN_AND_MEAN)
elseif(APPLE)
  if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
    set(CPLEX_PLATFORM arm64_osx CACHE STRING "CPLEX installed platform directory")
  else()
    set(CPLEX_PLATFORM x86-64_osx CACHE STRING "CPLEX installed platform directory")
  endif()
  set(_CPLEX_LIBRARY_LAYOUT static_pic)
else()
  set(CPLEX_PLATFORM x86-64_linux CACHE STRING "CPLEX installed platform directory")
  set(_CPLEX_LIBRARY_LAYOUT static_pic)
  find_package(OpenSSL REQUIRED COMPONENTS Crypto)
endif()
set(_CPLEX_LIBDIR "${CPLEX_ROOT}/cplex/lib/${CPLEX_PLATFORM}/${_CPLEX_LIBRARY_LAYOUT}")
set(_CONCERT_LIBDIR "${CPLEX_ROOT}/concert/lib/${CPLEX_PLATFORM}/${_CPLEX_LIBRARY_LAYOUT}")
if(WIN32 AND NOT cplex_LIBRARY)
  # The C callable library is cplex2211.lib/cplex2212.lib/etc., not cplex.lib.
  # Do not silently choose a different version when an installation has several.
  file(GLOB _CPLEX_VERSIONED_LIBS "${_CPLEX_LIBDIR}/cplex[0-9]*.lib")
  list(FILTER _CPLEX_VERSIONED_LIBS INCLUDE REGEX "/cplex[0-9]+\\.lib$")
  list(LENGTH _CPLEX_VERSIONED_LIBS _CPLEX_LIBRARY_COUNT)
  if(NOT _CPLEX_LIBRARY_COUNT EQUAL 1)
    message(FATAL_ERROR "Expected one versioned CPLEX library in ${_CPLEX_LIBDIR}; set cplex_LIBRARY to its exact .lib path")
  endif()
  list(GET _CPLEX_VERSIONED_LIBS 0 _CPLEX_LIBRARY)
  set(cplex_LIBRARY "${_CPLEX_LIBRARY}" CACHE FILEPATH "CPLEX callable library")
else()
  find_library(cplex_LIBRARY NAMES cplex HINTS "${_CPLEX_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
endif()
find_library(ilocplex_LIBRARY NAMES ilocplex HINTS "${_CPLEX_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
find_library(concert_LIBRARY NAMES concert HINTS "${_CONCERT_LIBDIR}" NO_DEFAULT_PATH REQUIRED)
message(STATUS "CPLEX platform: ${CPLEX_PLATFORM}; callable library: ${cplex_LIBRARY}")
add_library(mlnd_cplex INTERFACE)
target_include_directories(mlnd_cplex SYSTEM INTERFACE "${CPLEX_ROOT}/cplex/include" "${CPLEX_ROOT}/concert/include")
target_compile_definitions(mlnd_cplex INTERFACE IL_STD)
target_link_libraries(mlnd_cplex INTERFACE ${ilocplex_LIBRARY} ${concert_LIBRARY} ${cplex_LIBRARY})
if(NOT WIN32)
  find_package(Threads REQUIRED)
  target_link_libraries(mlnd_cplex INTERFACE m Threads::Threads ${CMAKE_DL_LIBS})
endif()
if(APPLE)
  target_link_libraries(mlnd_cplex INTERFACE "-framework CoreFoundation" "-framework IOKit" "-framework Accelerate")
endif()

add_library(mlnd_input STATIC src/input.cpp)
target_include_directories(mlnd_input PUBLIC include)
target_include_directories(mlnd_input SYSTEM PUBLIC "${NLOHMANN_JSON_INCLUDE_DIR}")
if(MSVC)
  target_compile_options(mlnd_input PRIVATE /W4)
  # Strict preserves NaN/Inf handling and disables contraction on all supported
  # MSVC versions. This does not promise bitwise equality across platforms.
  target_compile_options(mlnd_input PUBLIC /fp:strict /utf-8 /bigobj /EHsc)
else()
  target_compile_options(mlnd_input PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(mlnd_input PUBLIC -ffp-contract=off)
endif()
if(WIN32)
  target_link_libraries(mlnd_input PUBLIC bcrypt)
elseif(NOT APPLE)
  target_link_libraries(mlnd_input PUBLIC OpenSSL::Crypto)
endif()

add_executable(mlnd_input_probe EXCLUDE_FROM_ALL tests/input_probe.cpp)
target_link_libraries(mlnd_input_probe PRIVATE mlnd_input)
add_executable(mlnd_progress_probe EXCLUDE_FROM_ALL tests/progress_probe.cpp src/progress.cpp)
target_link_libraries(mlnd_progress_probe PRIVATE mlnd_input)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/path_catalogue.cpp")
  add_executable(mlnd_path_catalogue EXCLUDE_FROM_ALL tests/path_catalogue.cpp src/paths.cpp)
  target_link_libraries(mlnd_path_catalogue PRIVATE mlnd_input)
endif()

if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp")
  file(GLOB MLND_SOLVER_SOURCES CONFIGURE_DEPENDS src/*.cpp)
  list(REMOVE_ITEM MLND_SOLVER_SOURCES "${CMAKE_CURRENT_SOURCE_DIR}/src/input.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/main.cpp"
       "${CMAKE_CURRENT_SOURCE_DIR}/src/bdd.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/src/bdd_search.cpp")
  add_library(mlnd_solver STATIC ${MLND_SOLVER_SOURCES})
  target_link_libraries(mlnd_solver PUBLIC mlnd_input mlnd_cplex)
  if(MSVC)
    target_compile_options(mlnd_solver PRIVATE /W4)
  else()
    target_compile_options(mlnd_solver PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_executable(mlnd_cpp src/main.cpp)
  target_link_libraries(mlnd_cpp PRIVATE mlnd_solver)
  # Research diagnostics remain available explicitly; normal builds contain
  # the production solver only. Discover probe sources in one place.
  add_library(mlnd_bdd STATIC EXCLUDE_FROM_ALL src/bdd.cpp src/bdd_search.cpp)
  target_link_libraries(mlnd_bdd PUBLIC mlnd_solver)
  file(GLOB MLND_PROBE_SOURCES CONFIGURE_DEPENDS tests/*_probe.cpp)
  foreach(source ${MLND_PROBE_SOURCES})
    get_filename_component(probe ${source} NAME_WE)
    if(probe MATCHES "^(input|progress)_probe$")
      continue()
    endif()
    add_executable(mlnd_${probe} EXCLUDE_FROM_ALL ${source})
    if(probe MATCHES "^bdd")
      target_link_libraries(mlnd_${probe} PRIVATE mlnd_bdd)
    else()
      target_link_libraries(mlnd_${probe} PRIVATE mlnd_solver)
    endif()
  endforeach()
endif()
